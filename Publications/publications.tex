\section*{Publications included in this thesis}

\begin{enumerate}[label={[\arabic*]}, leftmargin=*, itemsep=4pt]
  \item W. Tong$^{*}$, \textbf{C. Sima}$^{*\dagger}$, T. Wang, L. Chen, S. Wu, H. Deng, Y. Gu, L. Lu, P. Luo, D. Lin, and H. Li$^{\dagger}$.
        ``Scene as Occupancy.''
        In \emph{IEEE/CVF International Conference on Computer Vision (ICCV)}, 2023.
        \emph{Role:} co-first author and co-project lead. The ICCV proceedings list W. Tong first; the arXiv version (arXiv:2306.02851) lists C. Sima first.
        \emph{Related challenges:} the author was the lead organiser of the CVPR 2023 3D Occupancy Prediction Challenge, which was co-hosted with the Occ3D team, used the Occ3D-nuScenes labels (0.4\,m voxels, 18 classes) rather than OpenOcc, and was ranked by voxel mIoU, and contributed the OpenOcc v2 benchmark (release v2.1; 0.4\,m voxels, 17 classes and occupancy flow, scored with RayIoU and mAVE) to the Occupancy and Flow track of the CVPR 2024 Autonomous Grand Challenge.
        \hfill (Chapter~\ref{ch:occnet})
  \item OpenScene Contributors.
        ``OpenScene: The Largest Up-to-Date 3D Occupancy Prediction Benchmark in Autonomous Driving.''
        Dataset release, \url{https://github.com/OpenDriveLab/OpenScene}, 2023.
        A redistribution of the nuPlan dataset at 2\,Hz with added occupancy and flow labels, released as data and a development kit without an accompanying paper; the official citation names the contributors collectively.
        \emph{Role:} project lead (with the OpenScene Contributors).
        \hfill (Chapter~\ref{ch:occnet})
  \item \textbf{C. Sima}$^{*}$, K. Renz$^{*}$, K. Chitta, L. Chen, H. Zhang, C. Xie, J. Bei{\ss}wenger, P. Luo, A. Geiger$^{\dagger}$, and H. Li$^{\dagger}$.
        ``DriveLM: Driving with Graph Visual Question Answering.''
        In \emph{European Conference on Computer Vision (ECCV)}, 2024. \textbf{Oral presentation.}
        \emph{Role:} co-first author. The author was also the lead organiser of the ``Driving with Language'' track of the CVPR 2024 Autonomous Grand Challenge, which was built on DriveLM and attracted 152 teams.
        \hfill (Chapter~\ref{ch:drivelm})
  \item \textbf{C. Sima}$^{*}$, K. Chitta$^{*}$, Z. Yu, S. Lan, P. Luo, A. Geiger, H. Li, and J. M. Alvarez.
        ``Centaur: Robust End-to-End Autonomous Driving with Test-Time Training.''
        \emph{arXiv preprint arXiv:2503.11650}, 2025.
        \emph{Role:} co-first author.
        \hfill (Chapter~\ref{ch:centaur})
  \item C. Yu, \textbf{C. Sima}$^{\dagger}$, G. Jiang, H. Zhang, H. Mai, H. Li, H. Wang, J. Chen, K. Wu, L. Chen$^{\dagger}$, L. Zhao, M. Shi, P. Luo, Q. Bu, S. Peng, T. Li, and Y. Yuan.
        ``$\chi_0$: Resource-Aware Robust Manipulation via Taming Distributional Inconsistencies.''
        \emph{arXiv preprint arXiv:2602.09021}, 2026.
        \emph{Role:} co-project lead, with L. Chen. The authors are listed in alphabetical order of first name.
        \hfill (Chapter~\ref{ch:kai0})
\end{enumerate}
